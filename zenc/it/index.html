import "std/net/tcp.zc"

fn main() {
    "Echo Server in ascolto su :8080";
    let listener = TcpListener::bind("127.0.0.1", 8080).unwrap();

    loop {
        // Accetta nuove connessioni
        let stream = listener.accept().unwrap();
        let buf: char[1024];
        
        while true {
            let n = stream.read(&buf[0], 1024).unwrap();
            if n == 0 { break; }
            stream.write(&buf[0], n);
        }
    }
}

struct Features {
    transparent_compilation: bool;
    native_c_abi:            bool;
    resource_management:     bool;
    monomorphization:        bool;
    pattern_matching:        bool;
    async_await:             bool;
}

impl Features {
    // Costruttore. Tutte le funzioni sono abilitate!
    fn new() -> Self {
        return Features{
            transparent_compilation: true,
            native_c_abi:            true,
            resource_management:     true,
            monomorphization:        true,
            pattern_matching:        true,
            async_await:             true,
        };
    }

    // Descrivi il linguaggio in una riga
    fn describe(self) {
        println "Ergonomia Moderna. Zero Overhead. Puro C.";
    }
}

fn main() {
    let zc = Features::new();
    zc.describe();
}
